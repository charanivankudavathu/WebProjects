SERVER-08

Create and use custom Node.js modules

Create a custom module and import it into another Node.js file.

Objective

To understand how custom modules can be created and reused in Node.js applications.

calculator.js

function add(a, b) {
    return a + b;
}

function subtract(a, b) {
    return a - b;
}

function multiply(a, b) {
    return a * b;
}

module.exports = {
    add,
    subtract,
    multiply
};

08.js

const calculator = require("./calculator");

console.log("Addition:", calculator.add(10, 5));
console.log("Subtraction:", calculator.subtract(10, 5));
console.log("Multiplication:", calculator.multiply(10, 5));

How to Run

node 08.js

Expected Output

Addition: 15
Subtraction: 5
Multiplication: 50
